Feature flags · v4.2

Ship on Friday. Flip it off on Saturday.

Kestrel puts every release behind a flag, evaluates it in 4 ms at the edge, and rolls it back without a deploy.

// checkout.ts
import { flag } from "@kestrel/edge";

if (await flag("new-checkout", user)) {
  return renderCheckoutV2();
}
